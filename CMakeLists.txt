# Isaac on any Native backend. `make` remains the local entry point; this
# file is what the platform SDKs build.
cmake_minimum_required(VERSION 3.20)
project(isaac LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(ISAAC_VERSION "0.1.0" CACHE STRING "Application version")
if(NOT NATIVE_ROOT)
    # Same dependency selection as Make/F5; -DNATIVE_ROOT=/path overrides it.
    execute_process(COMMAND sh "${CMAKE_SOURCE_DIR}/scripts/build/fetch_native.sh"
        OUTPUT_VARIABLE NATIVE_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE
        COMMAND_ERROR_IS_FATAL ANY)
endif()
set(BUILD_TESTING OFF CACHE BOOL "Native standalone tests" FORCE)
add_subdirectory(src/ui native-dependency)

set(libraries view impexp cmd book func expr val numfmt gfx util)
add_library(isaac_options INTERFACE)
target_include_directories(isaac_options INTERFACE include)
foreach(name ${libraries})
    target_include_directories(isaac_options INTERFACE "lib/${name}/include")
endforeach()
target_compile_options(isaac_options INTERFACE
    $<$<CXX_COMPILER_ID:GNU>:-Wall;-Wextra;-pedantic>)
target_compile_definitions(isaac_options INTERFACE
    ISAAC_VERSION="${ISAAC_VERSION}")
# Native's probe found std::jthread only behind libc++'s experimental
# library; its add_compile_options does not reach this directory.
if(APPLE AND NOT NATIVE_HAS_CXX20_THREADS)
    target_compile_options(isaac_options INTERFACE -fexperimental-library)
    target_link_options(isaac_options INTERFACE -fexperimental-library)
endif()

set(core_sources)
foreach(name ${libraries})
    file(GLOB sources CONFIGURE_DEPENDS "lib/${name}/*.cpp")
    list(APPEND core_sources ${sources})
endforeach()
add_library(isaac_core STATIC ${core_sources})
target_link_libraries(isaac_core PUBLIC isaac_options)

file(GLOB ui_sources CONFIGURE_DEPENDS src/ui/*.cpp src/app/*.cpp)
# Cross SDKs consume host-generated C++; never execute a target-built rc.
set(ISAAC_RESOURCE_CPP "" CACHE FILEPATH "Resource C++ generated by a host Native rc")
if(ISAAC_RESOURCE_CPP)
    if(NOT EXISTS "${ISAAC_RESOURCE_CPP}")
        message(FATAL_ERROR "ISAAC_RESOURCE_CPP does not exist")
    endif()
    set(resource_cpp "${ISAAC_RESOURCE_CPP}")
elseif(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Cross builds require ISAAC_RESOURCE_CPP; use scripts/build/platforms.py or generate it with make resources")
else()
    set(resource_cpp "${CMAKE_BINARY_DIR}/resources/isaac.cpp")
    file(GLOB resource_images CONFIGURE_DEPENDS src/resources/icons/*/*.png)
    add_custom_command(OUTPUT "${resource_cpp}"
        BYPRODUCTS "${CMAKE_BINARY_DIR}/resources/isaac.resources"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/resources"
        COMMAND $<TARGET_FILE:native_rc>
            "${CMAKE_SOURCE_DIR}/src/resources/isaac.json"
            -o "${CMAKE_BINARY_DIR}/resources/isaac.resources"
            --cpp "${resource_cpp}"
        DEPENDS native_rc src/resources/isaac.json ${resource_images}
        VERBATIM)
endif()
add_custom_target(isaac_resources DEPENDS "${resource_cpp}")
add_library(isaac_ui STATIC ${ui_sources} "${resource_cpp}")
target_include_directories(isaac_ui PUBLIC src/ui)
target_link_libraries(isaac_ui PUBLIC isaac_core native)

add_executable(isaac WIN32 MACOSX_BUNDLE src/main.cpp)
# The desktop icon ships beside the executable (GEMiX converts it with resgen).
configure_file(assets/icons/ISAAC.PNG "${CMAKE_BINARY_DIR}/bin/ISAAC.PNG" COPYONLY)
target_link_libraries(isaac PRIVATE isaac_ui)
if(APPLE)
    set_target_properties(isaac PROPERTIES
        MACOSX_BUNDLE_BUNDLE_NAME "Isaac"
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.wischner.isaac"
        MACOSX_BUNDLE_BUNDLE_VERSION "${ISAAC_VERSION}"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${ISAAC_VERSION}")
endif()

file(GLOB ui_tests CONFIGURE_DEPENDS tests/ui/test_*.cpp)
add_executable(test_desktop ${ui_tests})
target_include_directories(test_desktop PRIVATE tests/unit)
target_compile_definitions(test_desktop PRIVATE
    FIXTURES="${CMAKE_SOURCE_DIR}/tests/fixtures")
target_link_libraries(test_desktop PRIVATE isaac_ui)
